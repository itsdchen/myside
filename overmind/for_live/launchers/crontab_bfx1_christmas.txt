# Daily usermsgs: widen thresh at certain times of day that have higher vol

# 2x thresh at 9:29, to make sure traders that start at 9:30 start with widened thresh
# note: make sure these traders start at latest at 9:28. will catch the last 1min of traders ending at 9:30, but probably okay
29 9 * * 1-5 /home/ubuntu/.venvs/v1/bin/python /home/ubuntu/scripts/usermsg.py --strat ".*" --um SET_THRESH_MULT --args mult:2 >> /home/ubuntu/estrader/logs/um_logs_$(date +\%Y\%m\%d).log
# 1.5x thresh at 9:45
45 9 * * 1-5 /home/ubuntu/.venvs/v1/bin/python /home/ubuntu/scripts/usermsg.py --strat ".*" --um SET_THRESH_MULT --args mult:1.5 >> /home/ubuntu/estrader/logs/um_logs_$(date +\%Y\%m\%d).log
# back to 1x thresh at 10:00
00 10 * * 1-5 /home/ubuntu/.venvs/v1/bin/python /home/ubuntu/scripts/usermsg.py --strat ".*" --um SET_THRESH_MULT --args mult:1 >> /home/ubuntu/estrader/logs/um_logs_$(date +\%Y\%m\%d).log
# 2x thresh at 18:00:30. will miss the first 15s of trading for now.
# note: make sure these traders start at latest at 18:00.
00 18 * * 0-4 sleep 30; /home/ubuntu/.venvs/v1/bin/python /home/ubuntu/scripts/usermsg.py --strat ".*" --um SET_THRESH_MULT --args mult:2 >> /home/ubuntu/estrader/logs/um_logs_$(date +\%Y\%m\%d).log
# back to 1x thresh at 18:15
15 18 * * 0-4 /home/ubuntu/.venvs/v1/bin/python /home/ubuntu/scripts/usermsg.py --strat ".*" --um SET_THRESH_MULT --args mult:1 >> /home/ubuntu/estrader/logs/um_logs_$(date +\%Y\%m\%d).log
